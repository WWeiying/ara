cmake_minimum_required(VERSION 3.16)
project(output_residency_graph LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "riscv64")
    message(FATAL_ERROR "RISC-V Linux toolchain required")
endif()
if(NOT EXISTS "${SDK_ROOT}/software/akv/experimental/quantized_kv/output_tile/ggml_entry.h")
    message(FATAL_ERROR "Pinned SDK_ROOT required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_RVV ON CACHE BOOL "" FORCE)
set(GGML_RV_ZFH ON CACHE BOOL "" FORCE)
set(GGML_RV_ZVFH ON CACHE BOOL "" FORCE)
set(GGML_RV_ZICBOP OFF CACHE BOOL "" FORCE)
set(GGML_RV_ZIHINTPAUSE OFF CACHE BOOL "" FORCE)
set(GGML_RV_ZVFBFWMA OFF CACHE BOOL "" FORCE)
set(GGML_RISCV_QBS ON CACHE BOOL "" FORCE)
set(GGML_RISCV_AKV ON CACHE BOOL "" FORCE)
set(GGML_RISCV_AKV_QUANT_KV ON CACHE BOOL "" FORCE)
set(GGML_RISCV_AKV_REGISTER_PV ON CACHE BOOL "" FORCE)
set(GGML_RISCV_AKV_OUTPUT_TILE ON CACHE BOOL "" FORCE)
set(GGML_RISCV_QBS_RUNTIME_DIR "${SDK_ROOT}/software/qbs" CACHE PATH "" FORCE)
set(GGML_RISCV_AKV_RUNTIME_DIR "${SDK_ROOT}/software/akv" CACHE PATH "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_subdirectory("${LLAMA_SOURCE_DIR}/ggml" "${CMAKE_BINARY_DIR}/ggml" EXCLUDE_FROM_ALL)
add_executable(output-graph test.cpp)
target_compile_features(output-graph PRIVATE cxx_std_17)
target_compile_options(output-graph PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=fast -Wall -Wextra -Werror)
target_include_directories(output-graph PRIVATE "${LLAMA_SOURCE_DIR}/ggml/src" "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
target_link_options(output-graph PRIVATE -static)
target_link_libraries(output-graph PRIVATE ggml ggml-cpu)
