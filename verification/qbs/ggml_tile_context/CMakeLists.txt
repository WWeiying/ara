cmake_minimum_required(VERSION 3.16)
project(qbs_ggml_tile_context LANGUAGES C CXX)

set(LLAMA_SOURCE_DIR "" CACHE PATH "External llama.cpp checkout (read only)")
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/ggml/CMakeLists.txt")
    message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the QBS-enabled llama.cpp checkout")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "riscv64")
    message(FATAL_ERROR "This test requires the native RISC-V GNU/Linux toolchain")
endif()
get_filename_component(SIGNOFF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." REALPATH)
get_filename_component(QBS_RUNTIME "${SIGNOFF_ROOT}/software/qbs" REALPATH)
if(DEFINED GGML_RISCV_QBS_RUNTIME_DIR)
    get_filename_component(REQUESTED_RUNTIME "${GGML_RISCV_QBS_RUNTIME_DIR}" REALPATH)
    if(NOT REQUESTED_RUNTIME STREQUAL QBS_RUNTIME)
        message(FATAL_ERROR "Use this signoff worktree's software/qbs runtime")
    endif()
endif()
set(GGML_RISCV_QBS_RUNTIME_DIR "${QBS_RUNTIME}" CACHE PATH "Signoff runtime" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Private static GGML libraries")
set(GGML_NATIVE OFF CACHE BOOL "Cross compilation" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "Use GGML threadpool" FORCE)
set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "QBS repack traits" FORCE)
set(GGML_RVV ON CACHE BOOL "RVV activation quantization" FORCE)
# Match build-rv64-qbs-segmented-20261023, not GGML's optional ISA defaults.
set(GGML_RV_ZFH OFF CACHE BOOL "Baseline rv64gcv" FORCE)
set(GGML_RV_ZVFH OFF CACHE BOOL "Baseline rv64gcv" FORCE)
set(GGML_RV_ZICBOP OFF CACHE BOOL "Baseline rv64gcv" FORCE)
set(GGML_RV_ZIHINTPAUSE OFF CACHE BOOL "Baseline rv64gcv" FORCE)
set(GGML_RV_ZVFBFWMA OFF CACHE BOOL "Baseline rv64gcv" FORCE)
set(GGML_RISCV_QBS ON CACHE BOOL "Native QBS adapter" FORCE)
set(GGML_RISCV_AKV OFF CACHE BOOL "QBS-only sidecar" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "Static backend registration" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "Only build this sidecar" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "Only build this sidecar" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Build external sources only in the caller's private binary tree.
add_subdirectory("${LLAMA_SOURCE_DIR}/ggml" "${CMAKE_CURRENT_BINARY_DIR}/ggml" EXCLUDE_FROM_ALL)
add_executable(qbs-ggml-tile-context test.cpp)
set_target_properties(qbs-ggml-tile-context PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_features(qbs-ggml-tile-context PRIVATE cxx_std_17)
target_compile_options(qbs-ggml-tile-context PRIVATE
    -UNDEBUG -fno-fast-math -ffp-contract=off -fno-tree-vectorize -Wall -Wextra)
target_include_directories(qbs-ggml-tile-context SYSTEM PRIVATE
    "${LLAMA_SOURCE_DIR}/ggml/src"
    "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
target_include_directories(qbs-ggml-tile-context PRIVATE "${QBS_RUNTIME}/include")
target_link_libraries(qbs-ggml-tile-context PRIVATE ggml ggml-cpu)
