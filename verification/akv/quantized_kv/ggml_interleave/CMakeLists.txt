cmake_minimum_required(VERSION 3.16)
project(ggml_quant_kv_interleave LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "riscv64")
    message(FATAL_ERROR "Use the native RISC-V Linux toolchain")
endif()
get_filename_component(SIGNOFF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." REALPATH)
set(LLAMA_SOURCE_DIR "/home/wangwy/llama/llama.cpp" CACHE PATH "Private llama.cpp source")
set(GGML_ACCEPTED_BUILD "" CACHE PATH "Accepted live-graph static library build")
add_executable(ggml-quant-kv-interleave test.cpp)
target_compile_features(ggml-quant-kv-interleave PRIVATE cxx_std_17)
target_compile_definitions(ggml-quant-kv-interleave PRIVATE GGML_USE_RISCV_QUANT_KV)
target_compile_options(ggml-quant-kv-interleave PRIVATE -fno-fast-math -ffp-contract=off -Wall -Wextra -Werror)
target_include_directories(ggml-quant-kv-interleave SYSTEM PRIVATE
    "${LLAMA_SOURCE_DIR}/ggml/include" "${LLAMA_SOURCE_DIR}/ggml/src" "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cpu")
target_include_directories(ggml-quant-kv-interleave PRIVATE "${SIGNOFF_ROOT}/software/qbs/include")
foreach(name ggml ggml-cpu ggml-base)
    if(NOT EXISTS "${GGML_ACCEPTED_BUILD}/ggml/src/lib${name}.a")
        message(FATAL_ERROR "Missing accepted library ${name}")
    endif()
    target_link_libraries(ggml-quant-kv-interleave PRIVATE "${GGML_ACCEPTED_BUILD}/ggml/src/lib${name}.a")
endforeach()
target_link_libraries(ggml-quant-kv-interleave PRIVATE m dl pthread)
