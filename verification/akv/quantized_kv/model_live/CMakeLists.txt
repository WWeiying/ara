cmake_minimum_required(VERSION 3.16)
project(quant_kv_model_live LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "riscv64")
    message(FATAL_ERROR "Use the RISC-V Linux toolchain")
endif()
get_filename_component(SIGNOFF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." REALPATH)
set(LLAMA_SOURCE_DIR "/home/wangwy/llama/llama.cpp" CACHE PATH "Private llama.cpp checkout")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Private static libraries" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Cross compilation" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "GGML threadpool" FORCE)
set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "QBS weights" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "Static backend" FORCE)
set(GGML_RVV ON CACHE BOOL "Production RVV" FORCE)
set(GGML_RV_ZFH ON CACHE BOOL "Scalar half" FORCE)
set(GGML_RV_ZVFH ON CACHE BOOL "Vector half" FORCE)
set(GGML_RV_ZICBOP OFF CACHE BOOL "Baseline" FORCE)
set(GGML_RV_ZIHINTPAUSE OFF CACHE BOOL "Baseline" FORCE)
set(GGML_RV_ZVFBFWMA OFF CACHE BOOL "Baseline" FORCE)
set(GGML_RISCV_QBS ON CACHE BOOL "Native projection" FORCE)
set(GGML_RISCV_AKV ON CACHE BOOL "Existing Attention backend" FORCE)
set(GGML_RISCV_AKV_QUANT_KV ON CACHE BOOL "Opt-in quantized Attention under test" FORCE)
set(GGML_RISCV_QBS_RUNTIME_DIR "${SIGNOFF_ROOT}/software/qbs" CACHE PATH "QBS runtime" FORCE)
set(GGML_RISCV_AKV_RUNTIME_DIR "${SIGNOFF_ROOT}/software/akv" CACHE PATH "AKV runtime" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "Public model API only" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "Focused test only" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "Focused test only" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "Focused test only" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "Focused test only" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "Offline model" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "No network" FORCE)
set(LLAMA_HTTPLIB OFF CACHE BOOL "No server" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_executable(quant-kv-model-live test.cpp)
target_compile_features(quant-kv-model-live PRIVATE cxx_std_17)
target_compile_options(quant-kv-model-live PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=off -Wall -Wextra -Werror)
target_link_libraries(quant-kv-model-live PRIVATE llama)
