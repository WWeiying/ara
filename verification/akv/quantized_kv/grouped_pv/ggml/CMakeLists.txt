cmake_minimum_required(VERSION 3.16)
project(ggml_quant_kv_grouped_pv LANGUAGES C CXX)

if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "riscv64" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Use the GNU RISC-V Linux toolchain, not ZCC")
endif()
get_filename_component(SIGNOFF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." REALPATH)
set(LLAMA_SOURCE_DIR "/home/wangwy/llama/llama.cpp" CACHE PATH "Frozen private llama.cpp source")
set(GGML_RISCV_QBS_RUNTIME_DIR "${SIGNOFF_ROOT}/software/qbs" CACHE PATH "QBS runtime" FORCE)
set(GGML_RISCV_AKV_RUNTIME_DIR "${SIGNOFF_ROOT}/software/akv" CACHE PATH "AKV runtime" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Private static libraries" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Cross compilation" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "GGML threadpool" FORCE)
set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "QBS repack" FORCE)
set(GGML_RVV ON CACHE BOOL "Production RVV primitives" FORCE)
set(GGML_RV_ZFH ON CACHE BOOL "F16 scalar conversion" FORCE)
set(GGML_RV_ZVFH ON CACHE BOOL "Existing AKV assembly" FORCE)
set(GGML_RV_ZICBOP OFF CACHE BOOL "Baseline" FORCE)
set(GGML_RV_ZIHINTPAUSE OFF CACHE BOOL "Baseline" FORCE)
set(GGML_RV_ZVFBFWMA OFF CACHE BOOL "Baseline" FORCE)
set(GGML_RISCV_QBS ON CACHE BOOL "Native QBS" FORCE)
set(GGML_RISCV_AKV ON CACHE BOOL "AKV backend dispatch" FORCE)
set(GGML_RISCV_AKV_QUANT_KV ON CACHE BOOL "Quantized entry under test" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "Static registration" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "Focused test only" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "Focused test only" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_subdirectory("${LLAMA_SOURCE_DIR}/ggml" "${CMAKE_CURRENT_BINARY_DIR}/ggml" EXCLUDE_FROM_ALL)
add_executable(ggml-quant-kv-grouped-pv test.cpp)
set_target_properties(ggml-quant-kv-grouped-pv PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_features(ggml-quant-kv-grouped-pv PRIVATE cxx_std_17)
target_compile_options(ggml-quant-kv-grouped-pv PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=off -Wall -Wextra -Werror)
target_link_libraries(ggml-quant-kv-grouped-pv PRIVATE ggml ggml-cpu)
